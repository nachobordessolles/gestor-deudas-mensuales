async function syncToSheets(){ showSyncStatus('Guardando...'); try{ const payload={deudas,fijos,pagados,categorias,saldados}; const resp=await fetch(SHEET_URL,{ method:'POST', body:JSON.stringify(payload) }); const text=await resp.text(); if(text==='OK'){showSyncStatus('✓ Guardado',true);} else{showSyncStatus('✗ Error',false);} }catch(e){showSyncStatus('✗ Error',false);} }
